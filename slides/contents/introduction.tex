\section{任务背景与数据集}

\begin{frame}{任务背景与数据集}
  \small
  \begin{columns}[T]
    \column{0.50\textwidth}
    \textbf{任务目标}
    \begin{itemize}
      \setlength{\itemsep}{1pt}
      \item 识别孤立数字语音（0--9）
      \item 提取 MFCC 特征并完成分类
      \item 比较传统方法与深度学习方法
      \item 测试混合噪声下的鲁棒性
    \end{itemize}

    \vspace{3pt}
    \textbf{数据集}
    \begin{itemize}
      \setlength{\itemsep}{1pt}
      \item 7 位说话人，每人朗读 0--9
      \item 16 kHz，单声道 MP3
      \item 短时能量端点检测自动切分
    \end{itemize}

    \column{0.46\textwidth}
    \textbf{数据划分}
    \vspace{-2pt}
    \begin{table}
      \centering
      \footnotesize
      \begin{tabular}{lcc}
        \toprule
        划分 & 说话人 & 样本数 \\
        \midrule
        训练集 & speaker 1--5 & 50 \\
        验证集 & speaker 6--7 & 20 \\
        \bottomrule
      \end{tabular}
    \end{table}

    \vspace{-5pt}
    \textbf{评估设置}
    \begin{itemize}
      \setlength{\itemsep}{1pt}
      \item 纯净语音：准确率 + 混淆矩阵
      \item 带噪语音：7 级 SNR 鲁棒性
      \item SNR：$-10,-5,0,5,10,15,20$ dB
      \item 噪声：白噪声 + 1/f 噪声 + 50 Hz 工频
    \end{itemize}
  \end{columns}
\end{frame}
