\begin{frame}{MFCC 特征提取流程}
  \small
  \begin{columns}[T]
    \column{0.58\textwidth}
    \textbf{梅尔频率倒谱系数（MFCC）}
    \begin{enumerate}
      \setlength{\itemsep}{1pt}
      \item \textbf{预加重}：$y[n]=x[n]-0.97x[n-1]$
      \item \textbf{分帧加窗}：25 ms Hamming 窗，10 ms 帧移
      \item \textbf{FFT $\to$ 功率谱}：获得短时频谱能量
      \item \textbf{Mel 滤波}：模拟人耳非线性频率感知
      \item \textbf{Log + DCT}：压缩动态范围并去相关
      \item \textbf{动态特征}：在 13 维 MFCC 基础上加入 $\Delta$, $\Delta^2$，形成 39 维特征
    \end{enumerate}

    \vspace{0pt}
    {\footnotesize 注：C0 主要反映整体能量，数值较大，易掩盖其余倒谱系数的细节。因此 13 维 MFCC 图在可视化时省略 C0；39 维图保留完整动态特征结构。}
