\begin{frame}{MFCC 特征提取流程}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \textbf{梅尔频率倒谱系数（MFCC）}
    \begin{enumerate}
      \item \textbf{预加重}：$y[n] = x[n] - 0.97 x[n-1]$
      \item \textbf{分帧加窗}：25 ms Hamming 窗，10 ms 帧移
      \item \textbf{FFT $\to$ 功率谱}
      \item \textbf{Mel 滤波器组}（26 个三角滤波器，低频密、高频疏）
      \item \textbf{取对数}：压缩能量动态范围
      \item \textbf{DCT}：去相关，取前 13 维作为倒谱系数
      \item \textbf{动态特征}：$\Delta$ 和 $\Delta^2$（一阶/二阶差分）
    \end{enumerate}
