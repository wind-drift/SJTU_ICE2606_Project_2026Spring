\section{结果分析与总结}

\begin{frame}{纯净语音识别结果对比}
  \small
  \begin{columns}[T]
    \column{0.56\textwidth}
    \centering
    \textbf{纯净语音识别结果}
    \vspace{4pt}

    \begin{tabular}{lcc}
      \toprule
      \textbf{方法} & \textbf{维度} & \textbf{准确率} \\
      \midrule
      MFCC-stat + KNN   & 26 维 & 80.0\% \\
      MFCC+$\Delta$/$\Delta^2$+GMM & 39 维 & \alert{100.0\%} \\
      TinyKeywordCNN (102k) & 39 维 & 95.0\% \\
      \bottomrule
    \end{tabular}

    \vspace{8pt}
    \raggedright
    \begin{itemize}
      \setlength{\itemsep}{2pt}
      \item GMM：20 条验证样本全部识别正确
      \item CNN：1 个误判（$2\to6$）
      \item KNN：4 个混淆，说明统计特征丢失了部分时序信息
    \end{itemize}

    \column{0.38\textwidth}
    \textbf{对比结论}
    \begin{itemize}
      \setlength{\itemsep}{4pt}
      \item 动态特征能补充语音变化趋势
      \item GMM 保留变长序列信息，适合小样本任务
      \item CNN 具有端到端建模能力，但本实验验证集较小
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{总结与局限性}
  \small
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{已完成工作}
    \begin{itemize}
      \setlength{\itemsep}{2pt}
      \item 基于短时能量的自动端点检测
      \item 完整 MFCC 特征提取流程
      \item 传统方法：KNN 基线 + GMM 序列模型
      \item 深度方法：TinyKeywordCNN
      \item 7 级 SNR 混合噪声鲁棒性评估
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{核心结论}
    \begin{enumerate}
      \setlength{\itemsep}{2pt}
      \item MFCC 对孤立数字语音具有良好区分度
      \item GMM 在纯净语音与中高 SNR 下表现最好
      \item CNN 在低 SNR 下受益于数据增强，但单次测试统计意义有限
    \end{enumerate}

    \vspace{4pt}
    \textbf{局限性}
    \begin{itemize}
      \setlength{\itemsep}{1pt}
      \item 验证集仅 20 条，统计意义有限
      \item 只测试统一混合噪声，未拆分单类噪声影响
      \item 未进行说话人交叉验证
    \end{itemize}
  \end{columns}
\end{frame}
